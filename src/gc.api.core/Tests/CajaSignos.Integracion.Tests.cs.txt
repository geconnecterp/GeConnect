// Se ejecutan los servicios reales y se capturan los parámetros en la frontera SQL.
// No hay conexión a una base de datos ni llamadas al POS.
var checks = 0;
void Ok(bool condition, string message) { if (!condition) throw new Exception(message); checks++; }
var repository = DispatchProxy.Create<IRepository<EntidadBase>, SqlSimulado>();
var sql = (SqlSimulado)(object)repository;
var uow = new UowSimulado(repository);
var productos = new ApiProductoFactServicio(uow, new LogSimulado());
var pagos = new ApiPagoFactServicio(uow, new LogSimulado());
const string subtotal = """
[{"orden":1,"tipo":"SU","concepto":"Subtotal","base":0,"alicuota":0,"importe":82018.74,"id_aux":""},
 {"orden":2,"tipo":"CO","concepto":"Fernet 3x2 rico","base":0,"alicuota":0,"importe":-8201.87,"id_aux":"006787"}]
""";
const string producto = """
[{"p_id":"031225","cantidad_tot":0.250,"p_pvta_tot":82018.74,"cm_gravado":-67783.25,"cm_iva":-14235.49,"cm_dto":-8201.87}]
""";
sql.Calculo = new() { json_p = producto, json_subtotal = subtotal, json_sorteo = "[{\"so_sorteo\":\"0060\",\"so_desc\":\"Ejemplo\"}]" };
foreach (var filaUnica in new[] { false, true })
foreach (var tipo in new[] { "CR", "CF", "DP", "DV", "NC", "ND", "FS" })
{
    var etiqueta = $"{(filaUnica ? "CalcularFila" : "CalcularFilas")} / {tipo}";
    var calculoReq = new CalcularFilasReqDto { caja_id = "0001", adm_id = "0000", usu_id = "prueba", json_p = producto };
    var calculo = filaUnica ? productos.CalcularFila(calculoReq) : productos.CalcularFilas(calculoReq);
    Ok(sql.UltimoSp == (filaUnica ? "SPGECO_CAJA_Ope_Calcula_Fila" : "SPGECO_CAJA_Ope_Calcula_Filas"), etiqueta + ": SP de cálculo");
    Ok(calculo.json_subtotal == subtotal && calculo.json_p == producto, etiqueta + ": cálculo sin mutar");
    // Facturación pasa por DTO/sesión; NC y ND/NC/FS conservan el JSON del cálculo.
    var jsonSubtotal = tipo is "CR" or "CF" or "DP"
        ? JsonConvert.SerializeObject(JsonConvert.DeserializeObject<List<FactSubtotalJsonDto>>(calculo.json_subtotal))
        : calculo.json_subtotal;
    var request = CrearRequest(tipo, calculo.json_p, jsonSubtotal, calculo.json_sorteo);
    var respuesta = tipo == "DP" ? productos.CrearPagoDiferido(request) : pagos.ConfirmarOperacionCaja(request);
    Ok(respuesta.resultado == 0 && sql.UltimoSp == "SPGECO_CAJA_Ope_Confirmar", etiqueta + ": confirmación real con SQL simulado");
    var filas = JArray.Parse(sql.Parametros["@json_subtotal"]);
    Ok(filas[0]["importe"]!.Value<decimal>() == 82018.74m && filas[1]["importe"]!.Value<decimal>() == -8201.87m, etiqueta + ": importe SU/CO conserva signo");
    Ok(filas.Sum(f => f["importe"]!.Value<decimal>()) == 73816.87m, etiqueta + ": total 73816.87");
    Ok(filas[1]["id_aux"]!.Value<string>() == "006787", etiqueta + ": código sin alterar");
    var p = JArray.Parse(sql.Parametros["@json_p"])[0];
    Ok(p["cantidad_tot"]!.Value<decimal>() == 0.250m && p["cm_gravado"]!.Value<decimal>() == -67783.25m && p["cm_iva"]!.Value<decimal>() == -14235.49m && p["cm_dto"]!.Value<decimal>() == -8201.87m, etiqueta + ": productos e impuestos con signo");
    var valor = JArray.Parse(sql.Parametros["@json_valores"])[0];
    Ok(valor["rb_importe"]!.Value<decimal>() == 47.50m && valor["rb_rec"]!.Value<decimal>() == -5m && valor["rb_aux"]!.Value<decimal>() == -2.50m, etiqueta + ": Clover conserva total, porcentaje e importe del descuento");
    var credito = JArray.Parse(sql.Parametros["@json_union"])[0];
    Ok(credito["cv_importe"]!.Value<decimal>() == -30000m && credito["cv_importe_ori"]!.Value<decimal>() == -73515.35m, etiqueta + ": crédito aplicado y original con signo");
    Ok(credito["cv_fecha_carga"]!.Value<string>() == "2026-10-27 12:00:00.000", etiqueta + ": fecha no transformada");
    if (tipo == "CR" && !filaUnica && args.Length > 0)
    {
        Directory.CreateDirectory(args[0]);
        File.WriteAllText(Path.Combine(args[0], "json_subtotal-confirmacion.json"), filas.ToString(Formatting.Indented));
    }
}
foreach (var tipo in new[] { "CD", "CC", "CV", "IV" })
{
    var request = CrearRequest(tipo, "{}", "{}", "{}");
    request.json_valores = tipo == "CV"
        ? "[{\"ins_id\":\"CHE\",\"rb_importe\":20000},{\"ins_id\":\"PES\",\"rb_importe\":-20000}]"
        : "[{\"ins_id\":\"PES\",\"rb_importe\":100.00,\"rb_rec\":0,\"rb_aux\":0}]";
    pagos.ConfirmarOperacionCaja(request);
    var valores = JArray.Parse(sql.Parametros["@json_valores"]);
    Ok(valores[0]["rb_importe"]!.Value<decimal>() == (tipo == "CV" ? 20000m : 100m), tipo + ": ingreso positivo conservado");
    Ok(tipo != "CV" || valores[1]["rb_importe"]!.Value<decimal>() == -20000m, "CV: contrapartida negativa conservada");
    var cancela = JArray.Parse(sql.Parametros["@json_cancela"]);
    Ok(cancela[0]["cv_importe"]!.Value<decimal>() == 30000m && cancela[1]["cv_importe"]!.Value<decimal>() == -5000m, tipo + ": documentos con ambos signos");
    var union = JArray.Parse(sql.Parametros["@json_union"])[0];
    Ok(union["cv_importe"]!.Value<decimal>() == -30000m, tipo + ": NC no se convierte en deuda");
    Ok(sql.Parametros["@json_p"] == "{}" && sql.Parametros["@json_subtotal"] == "{}", tipo + ": objetos vacíos conservados");
}
foreach (var importe in new[] { -0.01m, 0m, 8201.87m })
{
    var request = CrearRequest("CR", "[]", JsonConvert.SerializeObject(new[] { new { orden = 1, tipo = "CO", importe } }), "[]");
    pagos.ConfirmarOperacionCaja(request);
    Ok(JArray.Parse(sql.Parametros["@json_subtotal"])[0]["importe"]!.Value<decimal>() == importe, "No inferir signo por tipo CO: " + importe);
}
Console.WriteLine($"PASS: {checks} verificaciones de cálculo/confirmación y parámetros SQL con signo. SQL/POS simulados.");

static CajaOpeConfirmarReq CrearRequest(string tipo, string productos, string subtotal, string sorteo) => new()
{
    co_tipo = tipo, caja_id = "0001", adm_id = "0000", usu_id = "prueba", caja_nro_proceso = "TEST-SIGNO", caja_nro_cierre = "1",
    json_p = productos, json_subtotal = subtotal, json_sorteo = sorteo,
    json_valores = "[{\"ins_id\":\"G00\",\"rb_importe\":47.50,\"rb_rec\":-5,\"rb_aux\":-2.50}]",
    json_union = "[{\"cv_importe\":\"-30000.00\",\"cv_importe_ori\":\"-73515,35\",\"cv_fecha_carga\":\"2026-10-27 12:00:00.000\"}]",
    json_cancela = "[{\"cv_importe\":30000},{\"cv_importe\":-5000}]"
};

public class SqlSimulado : DispatchProxy
{
    public string UltimoSp = "";
    public Dictionary<string, string> Parametros = new();
    public CalculaFilasResDto Calculo = new();
    protected override object? Invoke(MethodInfo? method, object?[]? arguments)
    {
        if (method?.Name != "EjecutarLstSpExt") throw new Exception("Acceso SQL inesperado: " + method?.Name);
        UltimoSp = (string)arguments![0]!;
        Parametros = ((List<SqlParameter>)arguments[1]!).Where(p => p.ParameterName.StartsWith("@json_")).ToDictionary(p => p.ParameterName, p => (string)p.Value);
        return UltimoSp switch
        {
            "SPGECO_CAJA_Ope_Calcula_Filas" or "SPGECO_CAJA_Ope_Calcula_Fila" => new List<CalculaFilasResDto> { Calculo },
            "SPGECO_CAJA_Ope_Confirmar" => new List<RespuestaDto> { new() { resultado = 0, resultado_msj = "OK simulado" } },
            _ => throw new Exception("SP inesperado: " + UltimoSp)
        };
    }
}
public class UowSimulado(IRepository<EntidadBase> repository) : IUnitOfWork
{
    public IRepository<T> GetRepository<T>() where T : EntidadBase => (IRepository<T>)repository;
    public int SaveChanges(bool process = true) => throw new NotSupportedException();
    public Task<int> SaveChangesAsync(bool process = true) => throw new NotSupportedException();
    public long Complete() => throw new NotSupportedException();
    public void InicializarTransaccion() => throw new NotSupportedException();
    public void Commit() => throw new NotSupportedException();
    public void Rollback() => throw new NotSupportedException();
}
public class LogSimulado : ILoggerHelper
{
    public Exception Log(Exception exception) => exception;
    public void Log(TraceEventType type, string message) { }
}
