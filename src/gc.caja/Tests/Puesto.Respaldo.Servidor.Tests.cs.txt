public class ProductoProxy:DispatchProxy {
 public List<ProductoDatosRequestDto> Requests=new();public bool Fallar;public bool NoExiste;
 protected override object? Invoke(MethodInfo? method,object?[]? args) {
  if(method?.Name!="ObtenerProductoDatos")throw new NotSupportedException();var r=(ProductoDatosRequestDto)args![0]!;Requests.Add(r);
  if(NoExiste)return Task.FromResult(new RespuestaGenerica<ProductoDatosResponseDto>{Ok=false,EsWarn=true,ListaEntidad=[]});
  return Task.FromResult(new RespuestaGenerica<ProductoDatosResponseDto>{Ok=!Fallar,ListaEntidad=[new(){p_id=r.valor,cantidad_tot=r.cantidad*12,respuesta=r.valor=="NO"?1:0,p_pvta=987,pre_id="viejo",cpf_nro="viejo"}]});
 }
}
