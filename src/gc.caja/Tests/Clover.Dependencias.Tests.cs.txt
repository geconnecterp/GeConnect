using System.Data;
using System.Diagnostics;
using gc.api.core.Contratos.Servicios.LineaCaja;
using gc.api.core.Servicios.LineaCaja;
using gc.api.infra.Datos;
using gc.api.infra.Extensions;
using gc.infraestructura.Core.Interfaces;
using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.DependencyInjection.Extensions;

// El registro y los servicios de producción se usan sin invocar repositorios ni SP.
var raiz = Path.Combine(Path.GetTempPath(), "geco-clover-di-" + Guid.NewGuid().ToString("N"));
Directory.CreateDirectory(raiz);
var casos = 0;
void Verificar(bool condicion, string nombre) { if (!condicion) throw new Exception(nombre); casos++; Console.WriteLine("OK " + nombre); }
WebApplicationBuilder Crear(string nombre)
{
    var contenido = Path.Combine(raiz, nombre); Directory.CreateDirectory(contenido);
    var b = WebApplication.CreateBuilder(new WebApplicationOptions { ContentRootPath = contenido, EnvironmentName = "Development", ApplicationName = typeof(Program).Assembly.GetName().Name });
    b.Configuration.Sources.Clear();
    b.Configuration.AddInMemoryCollection();
    b.Logging.ClearProviders();
    b.Host.UseDefaultServiceProvider(o => { o.ValidateOnBuild = true; o.ValidateScopes = true; });
    b.Services.AddDbContext<GeConnectContext>(o => o.UseSqlServer("Server=127.0.0.1,1;Database=SoloConstruccionDI;Integrated Security=True;TrustServerCertificate=True;Connect Timeout=1"));
    b.Services.AddServicios();
    b.Services.Replace(ServiceDescriptor.Scoped<ILoggerHelper, LogPruebaDI>());
    return b;
}
try
{
    // ApiPagoFactServicio ya no requiere el servicio/diario Clover para confirmar.
    var sinClover=Crear("confirmacion-independiente");sinClover.Services.RemoveAll<CloverServicio>();
    using(var host=sinClover.Build())using(var scope=host.Services.CreateScope())
        Verificar(scope.ServiceProvider.GetRequiredService<IApiPagoFactServicio>() is ApiPagoFactServicio,"Confirmación normal se construye sin CloverServicio ni estado POS");

    foreach (var nombre in new[] { "gc.api", "gc.api.report", "gc.api.gen" })
    {
        var b = Crear(nombre);
        var porDefecto = Path.Combine(b.Environment.ContentRootPath, "App_Data", "Clover");
        Verificar(!Directory.Exists(porDefecto), nombre + ": registrar servicios no crea el diario");
        Verificar(b.Services.Count(d => d.ServiceType == typeof(CloverServicio)) == 1, nombre + ": un registro Clover scoped");
        Verificar(b.Services.Count(d => d.ServiceType == typeof(CloverRegistro)) == 0, nombre + ": no registra diario del POS");
        using var app = b.Build();
        Verificar(!Directory.Exists(porDefecto), nombre + ": Build valida todas las dependencias sin acceder al diario");
        using var scope1 = app.Services.CreateScope();
        var pago = scope1.ServiceProvider.GetRequiredService<IApiPagoFactServicio>();
        var clover1 = scope1.ServiceProvider.GetRequiredService<CloverServicio>();
        Verificar(pago is ApiPagoFactServicio, nombre + ": construye ApiPagoFactServicio real");
        Verificar(ReferenceEquals(clover1, scope1.ServiceProvider.GetRequiredService<CloverServicio>()), nombre + ": conserva instancia Clover dentro del scope");
        Verificar(!Directory.Exists(porDefecto), nombre + ": resolver pago/Clover no accede al disco");
        using var scope2 = app.Services.CreateScope();
        Verificar(!ReferenceEquals(clover1, scope2.ServiceProvider.GetRequiredService<CloverServicio>()), nombre + ": scopes independientes");

        Verificar(scope1.ServiceProvider.GetRequiredService<GeConnectContext>().Database.GetDbConnection().State == ConnectionState.Closed, nombre + ": sin conexión SQL abierta");
    }
    {
        var b = Crear("registro-repetido"); b.Services.AddServicios();
        Verificar(b.Services.Count(d => d.ServiceType == typeof(CloverServicio)) == 1 && b.Services.Count(d => d.ServiceType == typeof(CloverRegistro)) == 0, "TryAdd evita duplicar servicios Clover");
        using var app = b.Build(); Verificar(true,"Registro repetido conserva grafo válido");
    }
    Console.WriteLine($"PASS: {casos} verificaciones de registro compartido, construcción, scopes y configuración. Sin SQL ni POS.");
}
finally
{
    var absoluta = Path.GetFullPath(raiz);
    if (absoluta.StartsWith(Path.GetFullPath(Path.GetTempPath()), StringComparison.OrdinalIgnoreCase) && Path.GetFileName(absoluta).StartsWith("geco-clover-di-")) Directory.Delete(absoluta, true);
}
public sealed class LogPruebaDI : ILoggerHelper { public Exception Log(Exception ex) => ex; public void Log(TraceEventType tipo, string mensaje) {} }
