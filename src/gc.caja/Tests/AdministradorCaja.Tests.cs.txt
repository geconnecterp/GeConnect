static void FakeSession(HttpContext context)
{
    context.User=new ClaimsPrincipal(new ClaimsIdentity(new[]{new Claim("AdmId","0007#Sucursal de prueba"),new Claim(ClaimTypes.Name,"operador-prueba")},"Test"));
    context.Session.SetString("Etiqueta","prueba");
    context.Session.SetString("CajaActual", "{\"CajaId\":\"0001\"}");
    var jwt=new JwtSecurityToken(claims:new[]{new Claim("user","operador-prueba")},expires:DateTime.UtcNow.AddHours(1));
    context.Request.Headers.Cookie = "prueba="+new JwtSecurityTokenHandler().WriteToken(jwt)+"; "+context.Request.Headers.Cookie;
}
class FakeApi : HttpMessageHandler
{
    public string Modo="vacio";
    public List<(string Method,string Url,string Body,string? Token)> Calls=new();
    protected override async Task<HttpResponseMessage> SendAsync(HttpRequestMessage r,CancellationToken ct)
    {
        Calls.Add((r.Method.Method,r.RequestUri!.ToString(),r.Content==null?"":await r.Content.ReadAsStringAsync(),r.Headers.Authorization?.Parameter));
        if (r.RequestUri!.AbsolutePath.EndsWith("ObtenerDatosCF") && Modo!="html" && Modo!="error")
        {
            var datos=new Dictionary<string,object> { ["caja_id"]=Modo=="puestodistinto"?"0099":"0001", ["adm_id"]=Modo=="sucursaldistinta"?"9999":"0007" };
            if(Modo!="sinestado") datos["caja_habilitadas"]=Modo=="estadoinvalido"?"X":
                new[]{"cerrada","sinresultado","sinproceso","incierto","warn","rechazo","posthtml","posterror"}.Contains(Modo)?"N":"S";
            return new(HttpStatusCode.OK){Content=new StringContent(JsonSerializer.Serialize(new{data=datos}))};
        }
        if (Modo=="posthtml" && r.Method==HttpMethod.Post) return new(HttpStatusCode.OK){Content=new StringContent("<html>Login</html>")};
        if ((Modo=="posterror" && r.Method==HttpMethod.Post) || Modo=="errorpuestos") return new(HttpStatusCode.ServiceUnavailable){Content=new StringContent("error")};
        if(Modo=="incierto" && r.Method==HttpMethod.Post) throw new HttpRequestException("Transporte simulado");
        if(Modo=="error") return new(HttpStatusCode.ServiceUnavailable){Content=new StringContent("error")};
        if(Modo=="html") return new(HttpStatusCode.OK){Content=new StringContent("<html>Login</html>")};
        string json=r.Method==HttpMethod.Get ? Modo switch {
            "sinlista"=>"{}", "nulo"=>"{\"data\":null}", "malpuesto"=>"{\"data\":[{}]}",
            "abiertos"=>"{\"data\":[{\"caja_id\":\"0001\",\"caja_nombre\":\"Puesto 1\",\"caja_nro_cierre\":12,\"usu_apellidoynombre\":\"Operador de prueba\"}]}",
            _=>"{\"data\":[]}" } : Modo switch {
            "sinresultado"=>"{\"data\":{\"resultado_msj\":\"OK\"}}",
            "sinproceso"=>"{\"data\":{\"resultado\":0}}",
            "warn"=>"{\"data\":{\"resultado\":2,\"resultado_msj\":\"No autorizado por SP\"}}",
            "rechazo"=>"{\"data\":{\"resultado\":-1,\"resultado_msj\":\"No autorizado por SP\"}}",
            _=>"{\"data\":{\"resultado\":0,\"resultado_id\":\"00-77777\",\"resultado_msj\":\"OK\"}}" };
        return new(HttpStatusCode.OK){Content=new StringContent(json)};
    }
}
