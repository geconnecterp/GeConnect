 public List<string> Mensajes=new();public IDisposable? BeginScope<TState>(TState state) where TState:notnull=>null;public bool IsEnabled(LogLevel level)=>true;
 public void Log<TState>(LogLevel level,EventId id,TState state,Exception? error,Func<TState,Exception?,string> formatter)=>Mensajes.Add(formatter(state,error));
}

public class NcCalculoPrueba:DispatchProxy
{
 public ApiProductoFactServicio Servicio=null!;public CalcularFilasReqDto? Request;
 protected override object? Invoke(MethodInfo? m,object?[]? args){
  if(m!.Name!="CalcularFilas")throw new Exception("Servicio NC inesperado; solo se permite calculo simulado: "+m.Name);
  Request=(CalcularFilasReqDto)args![0]!;
  return Task.FromResult(new RespuestaGenerica<CalculaFilasResDto>{Ok=true,Entidad=Servicio.CalcularFilas(Request),Mensaje="SP simulado"});
 }
}
