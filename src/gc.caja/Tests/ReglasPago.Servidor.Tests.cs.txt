// API local simulada: probar el servicio real sin consultar ni modificar la base de datos.
var builder = WebApplication.CreateBuilder();
builder.Logging.ClearProviders();
builder.WebHost.UseUrls("http://127.0.0.1:0");
await using var app = builder.Build();
string respuesta = "{\"data\":[]}";
app.MapPost("/api/apipagofactura/ObtenerFacturasPendientes", () => respuesta);
await app.StartAsync();
try {
    var servicio = new FactDiferidaServicio(Options.Create(new AppSettings { RutaBase = app.Urls.Single() }), NullLogger<FactDiferidaServicio>.Instance);
    var vacio = await servicio.ObtenerFacturasPendientes(new FactPendienteRequestDto(), "");
    Check(vacio.Ok && !vacio.EsError && !vacio.EsWarn && vacio.ListaEntidad?.Count == 0, "API sin facturas no es error");
    respuesta = "{\"data\":null}";
    var invalido = await servicio.ObtenerFacturasPendientes(new FactPendienteRequestDto(), "");
    Check(!invalido.Ok, "Respuesta nula no se confunde con lista vacía");
    Console.WriteLine("PASS: servicio real distingue pendientes vacíos de una respuesta inválida.");
} finally { await app.StopAsync(); }
